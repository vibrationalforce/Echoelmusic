   732      /// The same line read from the live process. `ActivePrewarm` is the environment flag iOS
   733      /// sets on a prewarmed launch; boot time and this process's start time are two `sysctl`s
   734      /// (`KERN_BOOTTIME`, `KERN_PROC_PID`); available memory is `os_proc_available_memory`
   735      /// (iOS only — the reader `MemoryPressureHandler` already uses); thermal state and Low Power
   736      /// Mode come from `ProcessInfo`. No actor, no closure handed to a framework, no I/O beyond
   737      /// the two sysctls: this runs inside the first stored property's initializer.
   738      static func launchContextLine() -> String {
   739          let info = ProcessInfo.processInfo
   740          let now = Date().timeIntervalSince1970
   741          let boot = bootTime()
   742          let started = processStartTime()
   743          #if os(iOS)
   744          let availableMB: Int? = Int(os_proc_available_memory()) / (1024 * 1024)
   745          #else
   746          let availableMB: Int? = nil   // `os_proc_available_memory` is iOS-family only
   747          #endif
   748          return launchContextLine(
   749              prewarm: info.environment["ActivePrewarm"] == "1",
   750              bootEpoch: boot.map { Int($0) },
   751              uptimeSeconds: boot.map { Int(now - $0) },
   752              processAgeMs: started.map { Int((now - $0) * 1000) },
   753              availableMB: availableMB,
   754              thermal: SelfCheckLine.thermalName(info.thermalState),
   755              lowPower: info.isLowPowerModeEnabled)
   756      }
