   760      private static func bootTime() -> TimeInterval? {
   761          #if canImport(Darwin)
   762          var mib: [Int32] = [CTL_KERN, KERN_BOOTTIME]
   763          var value = timeval()
   764          var size = MemoryLayout<timeval>.stride

    98      func testTheLiveReaderKeepsTheShape() throws {
    99          let live = EchoelCrashLog.launchContextLine()
   100          XCTAssertTrue(live.hasPrefix(EchoelCrashLog.launchContextPrefix), "the live line must carry the prefix: \(live)")
   101          XCTAssertFalse(live.contains(EchoelCrashLog.crashMarker), "the live line must not carry the crash marker: \(live)")
   102          XCTAssertEqual(live, live.lowercased(), "the live line is lower-case throughout: \(live)")
   103  
   104          let fields = live.dropFirst(EchoelCrashLog.launchContextPrefix.count)
   105              .split(separator: " ")
   106              .map { $0.split(separator: "=", maxSplits: 1).map(String.init) }
   107          XCTAssertEqual(fields.map { $0.first ?? "" },
   108                         ["prewarm", "boot", "up", "age", "avail", "thermal", "lowpower"],
   109                         "the live line names its seven fields in the fixed order: \(live)")
   110          let values = Dictionary(uniqueKeysWithValues: fields.compactMap { pair in
   111              pair.count == 2 ? (pair[0], pair[1]) : nil
   112          })
   113          // Always readable on any host: the environment and `ProcessInfo`.
   114          XCTAssertTrue(["0", "1"].contains(values["prewarm"] ?? ""), "`prewarm` is a flag, read from the environment: \(live)")
   115          XCTAssertTrue(["0", "1"].contains(values["lowpower"] ?? ""), "`lowpower` is a flag, read from `ProcessInfo`: \(live)")
   116          XCTAssertNotEqual(values["thermal"], "-", "`thermal` comes from `ProcessInfo` and is always named: \(live)")
   117          // Readable on this host only if the kernel answers: a dash is a stated refusal, a value is a number.
   118          for (name, unit) in [("boot", ""), ("up", "s"), ("age", "ms"), ("avail", "mb")] {
   119              let value = try XCTUnwrap(values[name], "field `\(name)` missing: \(live)")
   120              if value == "-" { continue }
   121              XCTAssertTrue(value.hasSuffix(unit), "`\(name)` carries its unit `\(unit)`: \(live)")
   122              XCTAssertNotNil(Int(value.dropLast(unit.count)), "`\(name)` is a whole number: \(live)")
   123          }
   124      }
