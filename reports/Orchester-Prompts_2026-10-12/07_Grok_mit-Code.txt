  157      /// `itemSeconds` the lane holds the item's last frame instead, exactly as a measured clip does.
  158      /// Required, never defaulted: a default no call site writes appears in no diff (#431).
  159      public static func resync(_ shot: Shot, observedSeconds: Double, itemSeconds: Double?) -> Resync {
  160          var expected = shot.mediaSeconds
  161          var still = shot.holdsLastFrame
  162          if !still, let held = lastFrame(at: expected, fileSeconds: itemSeconds) {
  163              expected = held
  164              still = true
  165          }
  166          guard observedSeconds.isFinite else { return .seek(toSeconds: expected) }
  167          let drift = observedSeconds - expected
  168          let size = abs(drift)
  169          if size <= deadbandSeconds { return .hold }
  170          if still || size > seekBeyondSeconds { return .seek(toSeconds: expected) }
  171          let slew = ((size - deadbandSeconds) * slewPerSecondOfDrift).clamped(to: 0...maxSlew)
  172          return .nudge(rate: Float(drift > 0 ? 1 - slew : 1 + slew))
  173      }

   53  /// "Frame `frame` of the master ring was rendered at host time `hostTicks`, at `sampleRate`."
   54  /// Written by the master tap (`RetroCapture`) once per buffer; read by the broadcast pump.
   55  struct BroadcastAudioAnchor: Sendable, Equatable {
   56      let frame: Int64
   57      let hostTicks: UInt64
   58      let sampleRate: Double
   59  
   60      var isUsable: Bool { sampleRate.isFinite && sampleRate > 0 && hostTicks > 0 && frame >= 0 }
   61  
   62      /// Host time at which ring frame `f` was rendered. Frames BEFORE the anchor are allowed —
   63      /// the ring holds 30 s of history behind the newest anchor.
   64      func hostTicks(forFrame f: Int64, timebase: BroadcastTimebase) -> UInt64? {
   65          guard isUsable else { return nil }
   66          let deltaSeconds = Double(f - frame) / sampleRate
   67          let deltaTicks = deltaSeconds * timebase.ticksPerSecond
   68          let result = Double(hostTicks) + deltaTicks
   69          guard result.isFinite, result > 0, result < Double(UInt64.max) else { return nil }
   70          return UInt64(result)
   71      }
   72  
   73      /// The ring frame rendered at host time `t` (rounded UP, so the frame returned is never
   74      /// earlier than `t` — the start gate must not send audio from before the picture's `T0`).
   75      func frame(atHostTicks t: UInt64, timebase: BroadcastTimebase) -> Int64? {
   76          guard isUsable else { return nil }
   77          let deltaSeconds = (Double(t) - Double(hostTicks)) / timebase.ticksPerSecond
   78          let deltaFrames = (deltaSeconds * sampleRate).rounded(.up)
   79          guard deltaFrames.isFinite, abs(deltaFrames) < 1e15 else { return nil }
   80          return frame + Int64(deltaFrames)
   81      }
